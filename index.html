// 每一幕都是异步任务：Promise + setTimeout 模拟等待
const delay = (ms) => new Promise(res => setTimeout(res, ms));

const playStage = async (s) => {
  await delay(s.duration);            // await 等待异步结果
  if (Math.random() < s.failRate) {
    throw new Error(s.failText);      // 随机失败 → 抛给 catch
  }
  return s.okText;
};

// async/await 串联整段寻宝旅程
async function startAdventure() {
  try {
    for (const s of STAGES) {
      const msg = await playStage(s);   // 逐幕等待，场景同步切换
      renderDone(s, msg);
    }
    celebrate();                          // 八幕全通，开启宝箱
  } catch (err) {
    renderFail(err.message);              // 失败 → 提供重试入口
  }
}
随机失败概率与 treasure2.js 保持一致（迷雾 30% · 鹦鹉 30% · 暗河 30% · 守卫 50% · 机关 30%）
